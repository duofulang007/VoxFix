cmake_minimum_required(VERSION 3.22)
project(voxfix_polish LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(voxfix-polish SHARED polish.cpp)
foreach(lib fst glog gflags_nothreads)
  add_library(wetext_${lib} STATIC IMPORTED)
  set_target_properties(wetext_${lib} PROPERTIES IMPORTED_LOCATION "${OPENFST_ANDROID}/jni/arm64-v8a/lib${lib}.a")
endforeach()
set_target_properties(wetext_fst PROPERTIES IMPORTED_LOCATION "${OPENFST_ANDROID}/jni/arm64-v8a/libfst-flat.a")
add_library(wetext_core STATIC vendor/wetext/processor/wetext_processor.cc vendor/wetext/processor/wetext_token_parser.cc vendor/wetext/utils/wetext_string.cc)
target_include_directories(wetext_core PUBLIC vendor/wetext "${OPENFST_ANDROID}/jni/include")
target_link_libraries(wetext_core PUBLIC wetext_fst wetext_glog wetext_gflags_nothreads log)
target_link_libraries(voxfix-polish PRIVATE llama wetext_core)
target_link_options(voxfix-polish PRIVATE "-Wl,-z,max-page-size=16384")
